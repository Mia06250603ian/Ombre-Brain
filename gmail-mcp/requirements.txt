# ============================================================
# gmail-mcp 依赖
#
# ⚠️ 每一条都钉上限。这不是洁癖，是 2026-07-29 那场事故的教训：
#    OB 的 requirements 写 `mcp>=1.0.0` 无上限，某次重建装到了当天刚发布的
#    mcp 2.0.0，而 2.0.0 删掉了 `mcp.server.fastmcp` 模块 →
#    启动即 ModuleNotFoundError → CrashLoopBackOff → Zeabur 挂起服务。
#    详见仓库根 requirements.txt 顶部和 OPERATIONS.md「OB 依赖钉版本」。
#
# 上限按「不许跨大版本」定，小版本和补丁照常升。
# 本文件成文时（2026-08-06）实际解析到的版本记在每行后面，
# 要升大版本请：改代码 + 跑单测 + 改这里，三件一起做。
#
# 邮件相关一个第三方库都没有：IMAP 用标准库 imaplib、邮件解析用标准库 email、
# HTML 转文本自己写（mailfilter.py）。依赖越少，这颗雷越小。
# ============================================================

# MCP 协议 SDK。上限必须钉死 <2.0.0，理由见上。
mcp>=1.0.0,<2.0.0          # 实测 1.29.0

# ASGI 服务器（server.py 直接 import 它跑 streamable-http）
uvicorn>=0.30.0,<1.0.0     # 实测 0.52.1

# Web 框架层：中间件（鉴权/CORS）和 JSONResponse 直接用到。
# 它本来也是 mcp 的传递依赖，但我们直接 import 了，就明写出来。
starlette>=0.37.0,<2.0.0   # 实测 1.4.1
